  // ---------------------------------------------------------------------------
  // 3. The auto-post form
  //
  //    A real <form> POST sends application/x-www-form-urlencoded, so the JSON
  //    travels as the value of one hidden field. The server reads it as a string
  //    and deserializes. This is the pattern to use when you need a genuine
  //    browser navigation POST (handoff to another app, no CORS preflight).
  // ---------------------------------------------------------------------------
  function autoPostForm(url, payload, target, popFeatures) {
    let windowName = target;
    let popup = null;


    // Open the popup FIRST, empty and named, then aim the form at that name.
    // Two reasons this order matters:
    //   1. window.open must run synchronously inside the click handler or the
    //      popup blocker kills it.
    //   2. Opening it blank avoids a wasted GET to the endpoint — the POST is
    //      what actually fills the window.

    popup = window.open(
      "",
      windowName,
      popFeatures
    );
    if (!popup || popup.closed) return { ok: false, blocked: true };
    // Something to look at while the POST is in flight.
    try {
      popup.document.write(
        "<!doctype html><title>Posting…</title>" +
        "<body style='font:15px system-ui;padding:40px;color:#555'>Submitting attachments…</body>"
      );
      popup.document.close();
    } catch (e) { /* cross-origin leftovers from a reused window — harmless */ }


    const form = document.createElement("form");
    form.name = "ClaimAIInsightsForm";   
    form.method = "POST";
    form.action = url;
    form.target = windowName;
    form.style.display = "none";

    // Setting .value as a property means no escaping problem — the browser
    // handles encoding when the form serializes.
    const field = document.createElement("input");
    field.type = "hidden";
    field.name = "payload";
    field.value = JSON.stringify(payload);
    form.appendChild(field);


    // Handy extras some endpoints expect alongside the blob:
    const idField = document.createElement("input");
    idField.type = "hidden";
    idField.name = "token";
    idField.value = "dev-token-0001";   //Temp value - to be replaced
    form.appendChild(idField);

    document.body.appendChild(form);
    console.log(popup.document.documentElement.outerHTML);

    form.submit();
    form.remove();          // the navigation is already in flight

    if (popup) popup.focus();
    return { ok: true, popup };
  }
